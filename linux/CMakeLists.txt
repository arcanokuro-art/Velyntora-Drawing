cmake_minimum_required(VERSION 3.22)
project(velyntora_linux LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(JNI REQUIRED)
find_package(Threads REQUIRED)
find_package(X11 REQUIRED COMPONENTS Xi)
add_subdirectory(../engine/core engine-build)
add_library(velyntora_desktop SHARED ../windows/native/DesktopBridge.cpp)
target_include_directories(velyntora_desktop PRIVATE ${JNI_INCLUDE_DIRS} ${X11_INCLUDE_DIR})
target_link_libraries(velyntora_desktop PRIVATE velyntora_engine Threads::Threads ${X11_LIBRARIES} ${X11_Xi_LIB})
target_link_options(velyntora_desktop PRIVATE -static-libgcc -static-libstdc++)
