cmake_minimum_required(VERSION 3.22)
project(velyntora_windows LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(JNI REQUIRED)
add_compile_options(-include ${CMAKE_CURRENT_SOURCE_DIR}/native/win_posix.hpp)
add_subdirectory(../engine/core engine-build)
add_library(velyntora_desktop SHARED native/DesktopBridge.cpp)
target_include_directories(velyntora_desktop PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(velyntora_desktop PRIVATE velyntora_engine)
target_link_options(velyntora_desktop PRIVATE -static-libgcc -static-libstdc++ -static)
set_target_properties(velyntora_desktop PROPERTIES PREFIX "" OUTPUT_NAME "velyntora_desktop")
